\chapter{Introduction}

\section{Purpose}
ThinkCraft Lab is a comprehensive web-based platform combining e-commerce functionality for 3D printing products with an integrated events management system for students and makers. The platform enables seamless product browsing, purchasing, order tracking, and community engagement through workshops, competitions, and courses.

\section{Intended Audience}
This SRS is intended for developers, project managers, QA teams, stakeholders, and end users (customers, students, administrators).

\section{Project Scope}
ThinkCraft Lab provides:
\begin{itemize}
\item \textbf{E-commerce}: 50+ products, shopping cart, wishlist, checkout with coupons, order management
\item \textbf{Events System}: Create/manage workshops, competitions, courses; registration tracking
\item \textbf{User Management}: Authentication, profiles, role-based access (Customer, Event Creator, Admin)
\item \textbf{Admin Dashboard}: Platform statistics, event management, user oversight
\end{itemize}

\chapter{Overall Description}

\section{Product Perspective}
ThinkCraft Lab is a standalone web application with client-server architecture:
\begin{itemize}
\item \textbf{Frontend}: React 18 + TypeScript + Vite + Tailwind CSS
\item \textbf{Backend}: Node.js + Express.js REST API
\item \textbf{Database}: MongoDB with Mongoose ODM
\item \textbf{Deployment}: Vercel (frontend), Render (backend), MongoDB Atlas
\end{itemize}

\section{User Classes}
\begin{itemize}
\item \textbf{Guest Users}: Browse products/events, view public pages
\item \textbf{Registered Customers}: Purchase products, register for events, manage profile
\item \textbf{Event Creators}: Create/manage events, view registrations
\item \textbf{Administrators}: Full system access, manage users/events, view analytics
\end{itemize}

\section{Operating Environment}
\textbf{Client}: Modern browsers (Chrome 90+, Firefox 88+, Safari 14+), responsive design (320px-2560px)\\
\textbf{Server}: Node.js 18+, MongoDB 5.0+, 512MB RAM minimum\\
\textbf{Production}: HTTPS, CDN-enabled, auto-scaling

\section{Constraints}
\begin{itemize}
\item Must use React, Node.js, Express, MongoDB, TypeScript
\item Session-based authentication with bcrypt password hashing
\item HTTPS required in production, CORS properly configured
\item Page load < 3s, API response < 500ms, support 100+ concurrent users
\end{itemize}

\chapter{Database Design}

\section{Collections Schema}

\textbf{Users}: \_id, username (unique), email (unique), password (hashed), isAdmin (boolean), createdAt

\textbf{Orders}: \_id, userId, orderNumber (unique), status, subtotal, discount, couponCode, totalAmount, shipping details, paymentMethod, items[], createdAt

\textbf{Events}: \_id, title, description, category, images[], date, location, isLive, createdBy, createdByUsername, registrations[{userId, username, registeredAt}], timestamps

\begin{itemize}
\item Users (1) → (Many) Orders
\item Users (1) → (Many) Events (creator)
\item Users (Many) ↔ (Many) Events (registrations)
\end{itemize}

\textbf{Orders}: userId, orderNumber (unique), createdAt\\
\textbf{Events}: category, isLive, date, createdBy

\chapter{Non-Functional Requirements}

\section{Performance}
\begin{itemize}
\item Page load < 3s, API response < 500ms (95\% requests)
\item Support 100+ concurrent users, 1000+ requests/min
\item Frontend bundle < 2MB, backend memory < 512MB
\item Optimized images, lazy loading, database indexes
\end{itemize}

\section{Security}
\begin{itemize}
\item Bcrypt password hashing (10 salt rounds)
\item Session-based auth, 7-day timeout
\item HTTPS encryption, secure cookies
\item Input validation (client \& server), XSS/CSRF protection
\item Role-based access control, environment variables for secrets
\end{itemize}

\section{Reliability}
\begin{itemize}
\item 99\% uptime, graceful error handling
\item User-friendly error messages, server-side logging
\item Database referential integrity, atomic transactions
\end{itemize}

\section{Usability}
\begin{itemize}
\item Registration < 2 min, purchase < 5 min
\item Intuitive navigation, responsive design (320px-2560px)
\item Keyboard navigable, WCAG 2.1 AA color contrast
\item Touch targets 44x44px minimum on mobile
\end{itemize}

\section{Maintainability}
\begin{itemize}
\item TypeScript for type safety, modular code structure
\item Inline comments, consistent naming conventions
\item API/database documentation, deployment guides
\item Unit/integration tests for critical functions
\end{itemize}

\chapter{API Endpoints}

\section{Authentication}
\begin{itemize}
\item POST /api/auth/signup - Register user
\item POST /api/auth/login - Authenticate user
\item POST /api/auth/logout - End session
\item GET /api/auth/me - Get current user
\end{itemize}

\section{Orders}
\begin{itemize}
\item POST /api/orders - Create order
\item GET /api/orders - Get user orders
\item GET /api/orders/:orderId - Get order details
\end{itemize}

\section{Events}
\begin{itemize}
\item GET /api/events - List events (query: category, live)
\item POST /api/events - Create event
\item GET /api/events/:eventId - Get event details
\item PUT /api/events/:eventId - Update event
\item DELETE /api/events/:eventId - Delete event
\item POST /api/events/:eventId/register - Register for event
\item DELETE /api/events/:eventId/register - Unregister
\end{itemize}

\section{User \& Admin}
\begin{itemize}
\item PUT /api/user/profile - Update profile
\item PUT /api/user/password - Change password
\item GET /api/admin/stats - Platform statistics
\end{itemize}

\chapter{Deployment}

\section{Production Environment}
\textbf{Frontend (Vercel)}:
\begin{itemize}
\item Build: npm run build, Output: dist/public
\item Env: VITE\_API\_URL=backend-url
\item Auto-deploy from GitHub, CDN-enabled
\end{itemize}

\textbf{Backend (Render)}:
\begin{itemize}
\item Build: npm install, Start: npm start
\item Env: MONGODB\_URI, SESSION\_SECRET, NODE\_ENV=production, FRONTEND\_URL
\item Auto-scaling, containerized deployment
\end{itemize}

\textbf{Database (MongoDB Atlas)}:
\begin{itemize}
\item M0 Free tier (512MB), replica sets
\item Network access: Allow all IPs for Render
\item Automatic backups, point-in-time recovery
\end{itemize}

\section{Maintenance}
\begin{itemize}
\item Monitor performance/uptime, review logs weekly
\item Update dependencies monthly, security patches promptly
\item Database backups weekly, archive old orders (2+ years)
\item Code refactoring, documentation updates
\end{itemize}
